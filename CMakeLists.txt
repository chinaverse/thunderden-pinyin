set(BIP39_WORDLIST "" CACHE FILEPATH "Pinned BIP39 English wordlist")
set(BIP39_WORDLIST_SHA256 "" CACHE STRING "Expected wordlist SHA256")

file(SHA256 "${BIP39_WORDLIST}" wordlist_hash)
if(NOT wordlist_hash STREQUAL BIP39_WORDLIST_SHA256)
  message(FATAL_ERROR "BIP39 wordlist hash mismatch")
endif()
file(STRINGS "${BIP39_WORDLIST}" words)
list(LENGTH words word_count)
if(NOT word_count EQUAL 2048)
  message(FATAL_ERROR "Expected 2048 BIP39 words")
endif()
set(word_array "#pragma once\n#include <array>\n#include <string_view>\nnamespace td { inline constexpr std::array<std::string_view, 2048> ENGLISH{\n")
foreach(word IN LISTS words)
  string(APPEND word_array "\"${word}\",\n")
endforeach()
string(APPEND word_array "}; }\n")
file(WRITE "${CMAKE_BINARY_DIR}/english.h" "${word_array}")

# Chinese recovery words are entered with pinyin, shown with GNU Unifont glyphs
# and mapped by wordlist index to the English words above before key derivation.
set(BIP39_CHINESE_WORDLIST "" CACHE FILEPATH "Pinned BIP39 Chinese (Simplified) wordlist")
set(BIP39_CHINESE_WORDLIST_SHA256 "" CACHE STRING "Expected Chinese wordlist SHA256")
set(UNIFONT_HEX "" CACHE FILEPATH "Pinned GNU Unifont .hex glyph file")
set(UNIFONT_HEX_SHA256 "" CACHE STRING "Expected Unifont .hex SHA256")
file(SHA256 "${BIP39_CHINESE_WORDLIST}" chinese_hash)
if(NOT chinese_hash STREQUAL BIP39_CHINESE_WORDLIST_SHA256)
  message(FATAL_ERROR "BIP39 Chinese wordlist hash mismatch")
endif()
file(SHA256 "${UNIFONT_HEX}" unifont_hash)
if(NOT unifont_hash STREQUAL UNIFONT_HEX_SHA256)
  message(FATAL_ERROR "Unifont glyph file hash mismatch")
endif()
file(STRINGS "${BIP39_CHINESE_WORDLIST}" chinese_words ENCODING UTF-8)
file(STRINGS "${CMAKE_SOURCE_DIR}/src/chinese_pinyin.txt" pinyin_lines ENCODING UTF-8 REGEX "^[^#]")
list(LENGTH chinese_words chinese_count)
list(LENGTH pinyin_lines pinyin_count)
if(NOT chinese_count EQUAL 2048 OR NOT pinyin_count EQUAL 2048)
  message(FATAL_ERROR "Expected 2048 Chinese words and pinyin entries")
endif()
file(STRINGS "${UNIFONT_HEX}" cjk_glyphs REGEX "^[4-9][0-9A-F][0-9A-F][0-9A-F]:[0-9A-F]+$")
foreach(line IN LISTS cjk_glyphs)
  string(SUBSTRING "${line}" 0 4 code)
  string(SUBSTRING "${line}" 5 -1 glyph_${code})
endforeach()
set(chinese_array "#pragma once\n#include <array>\n#include <cstdint>\n#include <string_view>\nnamespace td {\n")
string(APPEND chinese_array "inline constexpr std::array<std::string_view, 2048> CHINESE{\n")
set(pinyin_array "inline constexpr std::array<std::string_view, 2048> PINYIN{\n")
set(glyph_array "// 16x16 bitmaps, two bytes per row, most significant bit on the left.\ninline constexpr std::array<uint8_t, 2048 * 32> HANZI_GLYPHS{\n")
foreach(index RANGE 2047)
  list(GET chinese_words ${index} word)
  list(GET pinyin_lines ${index} entry)
  # The official list fixes each index; the pinyin column only selects candidates.
  if(NOT entry MATCHES "^([^ ]+) ([a-z]+( [a-z]+)*)$")
    message(FATAL_ERROR "Invalid pinyin entry ${index}")
  endif()
  set(readings "${CMAKE_MATCH_2}")
  if(NOT CMAKE_MATCH_1 STREQUAL word)
    message(FATAL_ERROR "Pinyin entry ${index} does not match the Chinese wordlist")
  endif()
  string(HEX "${word}" bytes)
  if(NOT bytes MATCHES "^(e[4-9])([89ab][0-9a-f])([89ab][0-9a-f])$")
    message(FATAL_ERROR "Chinese word ${index} is not one CJK character")
  endif()
  math(EXPR code "((0x${CMAKE_MATCH_1} & 0x0F) << 12) | ((0x${CMAKE_MATCH_2} & 0x3F) << 6) | (0x${CMAKE_MATCH_3} & 0x3F)"
    OUTPUT_FORMAT HEXADECIMAL)
  string(SUBSTRING "${code}" 2 -1 code)
  string(TOUPPER "${code}" code)
  if(NOT glyph_${code} MATCHES "^[0-9A-F]+$")
    message(FATAL_ERROR "Unifont has no glyph for Chinese word ${index}")
  endif()
  string(LENGTH "${glyph_${code}}" glyph_length)
  if(NOT glyph_length EQUAL 64)
    message(FATAL_ERROR "Unifont glyph for Chinese word ${index} is not 16x16")
  endif()
  string(REGEX REPLACE "(..)(..)(..)" "\\\\x\\1\\\\x\\2\\\\x\\3" escaped "${bytes}")
  string(APPEND chinese_array "\"${escaped}\",\n")
  string(APPEND pinyin_array "\"${readings}\",\n")
  string(REGEX REPLACE "([0-9A-F][0-9A-F])" "0x\\1," glyph_bytes "${glyph_${code}}")
  string(APPEND glyph_array "${glyph_bytes}\n")
endforeach()
string(APPEND chinese_array "};\n${pinyin_array}};\n${glyph_array}};\n}\n")
file(WRITE "${CMAKE_BINARY_DIR}/chinese.h" "${chinese_array}")
